import React, { useState, useEffect } from 'react';
import { useParams, useNavigate, Link } from 'react-router-dom';
import SiteHeader from '../../components/marketing/SiteHeader';
import { CreditCard, Lock } from 'lucide-react';

export default function CheckoutPage() {
  const { orderId } = useParams();
  const navigate = useNavigate();
  const [order, setOrder] = useState<any>(null);
  const [loading, setLoading] = useState(true);
  const [processing, setProcessing] = useState(false);
  const [error, setError] = useState('');
  const [cardNumber, setCardNumber] = useState('4242 4242 4242 4242');

  useEffect(() => {
    const fetchOrder = async () => {
      try {
        const res = await fetch(`${import.meta.env.VITE_API_URL || ''}/api/purchase/orders/${orderId}`);
        const data = await res.json();
        if (!res.ok) throw new Error(data.error || 'Order not found');
        setOrder(data);
      } catch (err: any) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };
    if (orderId) fetchOrder();
  }, [orderId]);

  const handlePayment = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    setProcessing(true);

    try {
      const cleanCard = cardNumber.replace(/\s/g, '');
      const res = await fetch(`${import.meta.env.VITE_API_URL || ''}/api/purchase/orders/${orderId}/pay`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ cardNumber: cleanCard }),
      });
      const data = await res.json();
      
      if (!res.ok) {
        throw new Error(data.error || 'Payment failed');
      }
      
      navigate(`/purchase/success/${orderId}`);
    } catch (err: any) {
      setError(err.message);
      setProcessing(false);
    }
  };

  if (loading) return <div className="min-h-screen bg-[#FAFAF7] grid place-items-center">Loading...</div>;
  if (!order) return <div className="min-h-screen bg-[#FAFAF7] grid place-items-center">{error}</div>;

  return (
    <div className="min-h-screen bg-[#FAFAF7]">
      <SiteHeader />
      <div className="pt-32 pb-24 px-6 md:px-12 max-w-lg mx-auto">
        <div className="text-center mb-8">
          <div className="w-16 h-16 bg-[#151C19] rounded-2xl flex items-center justify-center mx-auto mb-4">
            <Lock className="text-white" size={24} />
          </div>
          <h1 className="text-3xl font-bold text-[#151C19] mb-2">Secure Checkout</h1>
          <p className="text-[#151C19]/60">Total to pay: <strong className="text-[#151C19]">₹{order.amount}</strong></p>
        </div>

        <form onSubmit={handlePayment} className="bg-white p-8 rounded-3xl shadow-sm border border-gray-100">
          {error && <div className="mb-6 p-4 bg-rose-50 text-rose-600 rounded-xl text-sm font-medium">{error}</div>}
          
          <div className="mb-6 relative">
            <label className="block text-sm font-semibold text-[#151C19] mb-2">Card Number</label>
            <div className="relative">
              <input
                type="text"
                required
                value={cardNumber}
                onChange={e => setCardNumber(e.target.value)}
                className="w-full pl-12 pr-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:ring-2 focus:ring-[#E8783C]/20 focus:border-[#E8783C] transition-all font-mono"
                placeholder="0000 0000 0000 0000"
              />
              <CreditCard className="absolute left-4 top-1/2 -translate-y-1/2 text-gray-400" size={20} />
            </div>
            <p className="text-xs text-gray-500 mt-2">Use 4242 4242 4242 4242 for testing.</p>
          </div>

          <div className="grid grid-cols-2 gap-4 mb-8">
            <div>
              <label className="block text-sm font-semibold text-[#151C19] mb-2">Expiry Date</label>
              <input
                type="text"
                required
                defaultValue="12/28"
                className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:ring-2 focus:ring-[#E8783C]/20 focus:border-[#E8783C] transition-all font-mono"
                placeholder="MM/YY"
              />
            </div>
            <div>
              <label className="block text-sm font-semibold text-[#151C19] mb-2">CVC</label>
              <input
                type="text"
                required
                defaultValue="123"
                className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:ring-2 focus:ring-[#E8783C]/20 focus:border-[#E8783C] transition-all font-mono"
                placeholder="123"
              />
            </div>
          </div>

          <button
            type="submit"
            disabled={processing}
            className="w-full bg-[#151C19] hover:bg-[#151C19]/90 text-white py-4 rounded-xl font-bold transition-colors disabled:opacity-50 flex items-center justify-center gap-2"
          >
            {processing ? (
              <span className="flex items-center gap-2">Processing Payment...</span>
            ) : (
              <span className="flex items-center gap-2"><Lock size={18} /> Pay ₹{order.amount}</span>
            )}
          </button>
          <div className="mt-4 text-center">
            <Link to={`/purchase/review/${order.id}`} className="text-sm text-gray-500 hover:text-gray-800 underline">
              Cancel & Return
            </Link>
          </div>
        </form>
      </div>
    </div>
  );
}

    customerEmail: '',
    businessName: '',
  });

    e.preventDefault();

    setLoading(true);

    try {
      const res = await fetch(`${import.meta.env.VITE_API_URL || ''}/api/purchase/orders`, {
        method: 'POST',

          ...formData,
          planId: 'PREMIUM',

      setError(err.message);

  return (
